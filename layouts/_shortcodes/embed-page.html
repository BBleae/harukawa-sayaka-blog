{{- /* 嵌入一个独立页面：assets/embeds// 下的 body.html、style.css、script.js 用法：{{< embed-page "sessionless-architecture" >}} */ -}} {{- $name := .Get 0 -}} {{- $dir := printf "embeds/%s" $name -}} {{- with resources.Get (printf "%s/style.css" $dir) -}} {{- $css := . | minify | fingerprint -}} {{- end -}} {{- with resources.Get (printf "%s/body.html" $dir) -}} {{- .Content | safeHTML -}} {{- else -}} {{- errorf "embed-page: %s/body.html not found (%s)" $dir .Position -}} {{- end -}} {{- with resources.Get (printf "%s/script.js" $dir) -}} {{- $js := . | minify | fingerprint -}} {{- end -}}
